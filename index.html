function commitScore(team, customPlayer = null) { const fx=getActiveFx(); if(!fx || !stagedScore) return; let chosenPlayer = customPlayer; if(chosenPlayer === null) { chosenPlayer = document.getElementById('player-select-' + team).value; } let typeText = scoreTypeMap[String(stagedScore.pts)] || (stagedScore.pts + ' pts'); if(stagedScore.isKick) { if(stagedScore.kickSuccessful) { // Add points only if successful if(team === 1) fx.score1 = Math.max(0, (fx.score1 || 0) + stagedScore.pts); else fx.score2 = Math.max(0, (fx.score2 || 0) + stagedScore.pts); // FIXED: Removed the space between $ and { typeText = `${typeText} ✅ (+${stagedScore.pts})`; } else { // Missed kick - 0 points added typeText = `Missed ${typeText} ❌`; } } else { // Regular points (Try / Deductions) if(team === 1) fx.score1 = Math.max(0, (fx.score1 || 0) + stagedScore.pts); else fx.score2 = Math.max(0, (fx.score2 || 0) + stagedScore.pts); // FIXED: Removed the space between $ and { if(stagedScore.pts > 0) typeText = `${typeText} (+${stagedScore.pts})`; } const teamName = team === 1 ? fx.team1 : fx.team2; if(!fx.events) fx.events = []; fx.events.push({ team: teamName, type: typeText, time: fxTime(fx), player: chosenPlayer || "" }); document.getElementById('player-box-1').style.display = 'none'; document.getElementById('player-box-2').style.display = 'none'; stagedScore = null; db.ref('fixtures/' + fx.id).update(fx); }
